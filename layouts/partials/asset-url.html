{{/* Devuelve la URL de un archivo de static/ con ?v=. La URL solo cambia cuando cambia el archivo: ningún navegador ni proxy puede servir una versión vieja tras un despliegue. Uso: {{ partial "asset-url" "/app.css" }} */}} {{- $ruta := . -}} {{- $archivo := printf "static%s" $ruta -}} {{- $url := $ruta -}} {{- if os.FileExists $archivo -}} {{- $url = printf "%s?v=%s" $ruta (substr (os.ReadFile $archivo | md5) 0 10) -}} {{- end -}} {{- return $url -}}
